-- ===================== REQUEST HELPER =====================
local function GetRequestFunction()
    return request or http_request or (syn and syn.request)
end

local function DoRequest(url, method, body, headers)
    local requestFunction = GetRequestFunction()
    if not requestFunction then
        return nil
    end

    local success, result = pcall(function()
        return requestFunction({
            Url = url,
            Method = method or "GET",
            Headers = headers or {
                ["Cache-Control"] = "no-cache"
            },
            Body = body
        })
    end)

    if not success or not result then
        return nil
    end

    return result
end

-- ===================== DISCORD WEBHOOK =====================
local function SendToDiscord(data)
    local placeId = tostring(data.PlaceId or "?")
    local jobId = tostring(data.JobId or "?")
    local brainrots = data.Brainrots or {}

    local brainrotText = "Aucun"
    if type(brainrots) == "table" and #brainrots > 0 then
        brainrotText = table.concat(brainrots, ", ")
    elseif type(brainrots) == "string" and brainrots ~= "" then
        brainrotText = brainrots
    end

    local joinLink = string.format(
        "https://www.roblox.com/games/%s?jobId=%s",
        placeId,
        jobId
    )

    local embed = {
        title = "🔔 Nouveau Log détecté",
        color = 5763719, -- vert
        fields = {
            {
                name = "🧠 Brainrot",
                value = "```" .. brainrotText .. "```",
                inline = false
            },
            {
                name = "📍 PlaceId",
                value = "`" .. placeId .. "`",
                inline = true
            },
            {
                name = "🆔 JobId",
                value = "`" .. jobId .. "`",
                inline = true
            },
            {
                name = "🔗 Lien Join",
                value = "[Clique ici pour rejoindre](" .. joinLink .. ")",
                inline = false
            }
        },
        footer = {
            text = "LOG | NOTIFIER • ID #" .. tostring(data.id or "?")
        },
        timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ")
    }

    local payload = HttpService:JSONEncode({
        username = "Log Notifier",
        embeds = { embed }
    })

    task.spawn(function()
        DoRequest(WEBHOOK_URL, "POST", payload, {
            ["Content-Type"] = "application/json"
        })
    end)
end

-- ===================== LOOP =====================
task.spawn(function()
    while true do
        task.wait(3)

        if enabled then
            local result = DoRequest(API_URL, "GET")

            if result and result.Body then
                local success, data = pcall(function()
                    return HttpService:JSONDecode(result.Body)
                end)

                if success and data and data.payload then
                    if tostring(data.id) ~= tostring(lastLogId) then
                        lastLogId = data.id

                        local payload = data.payload

                        lastPlaceId = payload.PlaceId
                        lastJobId = payload.JobId

                        local logData = {
                            id = data.id,
                            PlaceId = payload.PlaceId,
                            JobId = payload.JobId,
                            Brainrots = payload.Brainrots
                        }

                        AddLog(logData)
                        SendToDiscord(logData) -- envoie sur Discord
                    end
                end
            end
        end
    end
end)
